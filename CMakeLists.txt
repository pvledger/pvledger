cmake_minimum_required(VERSION 3.18)
project(PVLedger VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(FetchContent)
find_package(oatpp 1.3.0 EXACT QUIET CONFIG)
if(NOT oatpp_FOUND)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  set(OATPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(oatpp GIT_REPOSITORY https://github.com/oatpp/oatpp.git GIT_TAG 1.3.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(oatpp)
  add_library(oatpp::oatpp ALIAS oatpp)
endif()
find_package(yaml-cpp 0.8 QUIET CONFIG)
if(NOT yaml-cpp_FOUND)
  set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(yaml-cpp GIT_REPOSITORY https://github.com/jbeder/yaml-cpp.git GIT_TAG 0.8.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(yaml-cpp)
  # yaml-cpp 0.8.0 relies on a transitive cstdint include on older compilers.
  target_compile_options(yaml-cpp PRIVATE -include cstdint)
endif()
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
add_executable(vpledger src/main.cpp src/config.cpp src/storage.cpp src/application.cpp)
target_compile_features(vpledger PRIVATE cxx_std_17)
target_compile_options(vpledger PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(vpledger PRIVATE src/headers)
file(RELATIVE_PATH runtime_data "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger")
target_compile_definitions(vpledger PRIVATE PVL_VERSION="${PROJECT_VERSION}" PVL_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}" PVL_DATA="${runtime_data}")
target_link_libraries(vpledger PRIVATE oatpp::oatpp yaml-cpp::yaml-cpp SQLite::SQLite3 OpenSSL::Crypto PkgConfig::JSONCPP)
install(TARGETS vpledger RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY templates js lang themes DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger)
install(FILES db/pvledger.sql DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger/db)
install(DIRECTORY src/plugins/ DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger/plugins)
file(READ src/config/config.yaml installed_config)
string(REPLACE "../../lang" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger/lang" installed_config "${installed_config}")
string(REPLACE "../../themes" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger/themes" installed_config "${installed_config}")
string(REPLACE "../plugins" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger/plugins" installed_config "${installed_config}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.example.yaml" "${installed_config}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/config.example.yaml" DESTINATION ${CMAKE_INSTALL_DOCDIR})
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli_help COMMAND vpledger --help)
  add_test(NAME cli_version COMMAND vpledger --version)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:vpledger>)
endif()
