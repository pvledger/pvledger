cmake_minimum_required(VERSION 3.18)
project(PVLedger VERSION 0.1.2 LANGUAGES C CXX)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  if(WIN32 AND CMAKE_HOST_WIN32)
    set(CMAKE_INSTALL_PREFIX "C:/pvledger" CACHE PATH "Installation prefix" FORCE)
  elseif(WIN32)
    set(CMAKE_INSTALL_PREFIX "${CMAKE_CURRENT_BINARY_DIR}/windows-install" CACHE PATH "Installation prefix" FORCE)
  else()
    set(CMAKE_INSTALL_PREFIX "/opt/pvledger" CACHE PATH "Installation prefix" FORCE)
  endif()
endif()
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM|arm|aarch64)" OR
     (MSVC AND NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64"))
    message(FATAL_ERROR "PVLedger Windows releases target x64")
  endif()
  enable_language(RC)
  set(CMAKE_INSTALL_BINDIR ".")
  set(CMAKE_INSTALL_DATADIR "share")
  set(CMAKE_INSTALL_DOCDIR "share/pvledger")
  set(static_default OFF)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(OATPP_MSVC_LINK_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
  set(OPENSSL_USE_STATIC_LIBS TRUE)
  set(OPENSSL_MSVC_STATIC_RT TRUE)
  set(VCPKG_APPLOCAL_DEPS OFF)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  if(MSVC)
    add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>" "$<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>")
  endif()
else()
  set(static_default ON)
endif()
include(GNUInstallDirs)
option(PVLEDGER_STATIC "Build a self-contained Linux release executable" ${static_default})
if(PVLEDGER_STATIC)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The static release target currently supports Linux")
  endif()
  set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()
if(PVLEDGER_STATIC OR WIN32)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build shared core libraries" FORCE)
  set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "Build shared yaml-cpp" FORCE)
endif()
find_package(oatpp 1.3.0 EXACT QUIET CONFIG)
if(NOT oatpp_FOUND)
  set(OATPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(oatpp GIT_REPOSITORY https://github.com/oatpp/oatpp.git GIT_TAG 1.3.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(oatpp)
  add_library(oatpp::oatpp ALIAS oatpp)
  if(MINGW AND NOT CMAKE_HOST_WIN32)
    # Oat++ 1.3.0 spells these SDK headers with Windows-specific casing.
    set(oatpp_mingw_headers "${CMAKE_CURRENT_BINARY_DIR}/oatpp-mingw-headers")
    file(WRITE "${oatpp_mingw_headers}/WinSock2.h" "#include <winsock2.h>\n")
    file(WRITE "${oatpp_mingw_headers}/Winsock2.h" "#include <winsock2.h>\n")
    file(WRITE "${oatpp_mingw_headers}/WS2tcpip.h" "#include <ws2tcpip.h>\n")
    target_include_directories(oatpp PRIVATE "${oatpp_mingw_headers}")
  endif()
endif()
find_package(yaml-cpp 0.8 QUIET CONFIG)
if(NOT yaml-cpp_FOUND)
  set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(yaml-cpp GIT_REPOSITORY https://github.com/jbeder/yaml-cpp.git GIT_TAG 0.8.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(yaml-cpp)
  # yaml-cpp 0.8.0 relies on a transitive cstdint include on older compilers.
  if(MSVC)
    target_compile_options(yaml-cpp PRIVATE /FIcstdint)
  else()
    target_compile_options(yaml-cpp PRIVATE -include cstdint)
  endif()
endif()
find_package(SQLite3 3.35 REQUIRED)
find_package(OpenSSL REQUIRED)
if(WIN32)
  find_package(jsoncpp CONFIG REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
endif()
if(PVLEDGER_STATIC)
  find_library(PVL_SQLITE_ARCHIVE NAMES libsqlite3.a REQUIRED)
  find_library(PVL_CRYPTO_ARCHIVE NAMES libcrypto.a REQUIRED)
  find_library(PVL_JSON_ARCHIVE NAMES libjsoncpp.a)
  if(PVL_JSON_ARCHIVE)
    set(json_library "${PVL_JSON_ARCHIVE}")
  else()
    set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
    set(JSONCPP_WITH_POST_BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
    set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
    FetchContent_Declare(jsoncpp GIT_REPOSITORY https://github.com/open-source-parsers/jsoncpp.git GIT_TAG 1.9.5 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(jsoncpp)
    set(json_library jsoncpp_static)
  endif()
  set(core_libraries ${PVL_SQLITE_ARCHIVE} ${PVL_CRYPTO_ARCHIVE} ${json_library} ${CMAKE_DL_LIBS})
elseif(WIN32)
  set(core_libraries SQLite::SQLite3 OpenSSL::Crypto JsonCpp::JsonCpp)
else()
  set(core_libraries SQLite::SQLite3 OpenSSL::Crypto PkgConfig::JSONCPP)
endif()
FetchContent_Declare(miniz GIT_REPOSITORY https://github.com/richgel999/miniz.git GIT_TAG 3.0.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(miniz)
add_executable(pvledger src/main.cpp src/config.cpp src/storage.cpp src/application.cpp src/portability.cpp src/spreadsheet.cpp)
set_target_properties(pvledger PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
if(MSVC)
  target_compile_options(pvledger PRIVATE /W4)
else()
  target_compile_options(pvledger PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_include_directories(pvledger PRIVATE src/headers ${SQLite3_INCLUDE_DIRS} ${OPENSSL_INCLUDE_DIR} ${JSONCPP_INCLUDE_DIRS})
if(WIN32)
  set(runtime_data "share/pvledger")
else()
  file(RELATIVE_PATH runtime_data "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger")
endif()
target_compile_definitions(pvledger PRIVATE PVL_VERSION="${PROJECT_VERSION}" PVL_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}" PVL_DATA="${runtime_data}")
target_link_libraries(pvledger PRIVATE oatpp::oatpp yaml-cpp::yaml-cpp miniz ${core_libraries})
if(WIN32)
  add_executable(pvledger-win-helper WIN32 windows/helper.cpp windows/helper.rc src/config.cpp)
  set_target_properties(pvledger-win-helper PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
  target_include_directories(pvledger-win-helper PRIVATE src/headers)
  target_compile_definitions(pvledger-win-helper PRIVATE
    PVL_VERSION="${PROJECT_VERSION}" PVL_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}" PVL_DATA="${runtime_data}")
  target_link_libraries(pvledger-win-helper PRIVATE
    yaml-cpp::yaml-cpp JsonCpp::JsonCpp OpenSSL::Crypto shell32 ole32 uuid)
  if(MINGW)
    target_link_options(pvledger PRIVATE -static -municode)
    target_link_options(pvledger-win-helper PRIVATE -static -municode)
  endif()
  install(TARGETS pvledger-win-helper RUNTIME DESTINATION . COMPONENT PVLedger)
endif()
if(PVLEDGER_STATIC)
  target_link_options(pvledger PRIVATE -static)
  find_program(PVL_READELF readelf REQUIRED)
  add_custom_command(TARGET pvledger POST_BUILD
    COMMAND ${CMAKE_COMMAND} -DBINARY=$<TARGET_FILE:pvledger> -DREADELF=${PVL_READELF}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/VerifyStatic.cmake)
endif()
install(TARGETS pvledger RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT PVLedger)
install(DIRECTORY templates js lang themes DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger COMPONENT PVLedger)
install(FILES db/pvledger.sql DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger/db COMPONENT PVLedger)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger COMPONENT PVLedger)
install(FILES ${miniz_SOURCE_DIR}/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger/licenses RENAME miniz.txt COMPONENT PVLedger)
install(DIRECTORY src/plugins/ DESTINATION ${CMAKE_INSTALL_DATADIR}/pvledger/plugins COMPONENT PVLedger)
if(WIN32)
  include(cmake/WindowsPackaging.cmake)
else()
  file(READ src/config/config.yaml installed_config)
  string(REPLACE "../../lang" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger/lang" installed_config "${installed_config}")
  string(REPLACE "../../themes" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger/themes" installed_config "${installed_config}")
  string(REPLACE "../plugins" "${CMAKE_INSTALL_FULL_DATADIR}/pvledger/plugins" installed_config "${installed_config}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.example.yaml" "${installed_config}")
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/config.example.yaml" DESTINATION ${CMAKE_INSTALL_DOCDIR} COMPONENT PVLedger)
  include(cmake/Packaging.cmake)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli_help COMMAND pvledger --help)
  add_test(NAME cli_version COMMAND pvledger --version)
  find_program(PVL_TEST_BROWSER NAMES chromium chromium-browser google-chrome brave-browser brave-origin)
  if(PVL_TEST_BROWSER)
    add_test(NAME import_formats COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/browser_formats.py ${PVL_TEST_BROWSER})
  endif()
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:pvledger>)
  if(NOT WIN32)
    add_test(NAME service COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/service.py
      $<TARGET_FILE:pvledger> ${CMAKE_CURRENT_BINARY_DIR}/packaging/pvledger.init)
    set_tests_properties(service PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
